import {it} from 'vitest';import {writeFileSync} from 'node:fs';import {World} from '../src/core/world';import {World as OldWorld} from '../build/qa/Stage-196/baseline-world.mjs';import {C} from '../src/core/config';import {safeRoute} from '../src/core/reachability';
it('records exact old/new route-forecast failures in the heavy chapter',()=>{const result=[];for(const [version,Runtime]of [['old',OldWorld],['current',World]]){const w=new Runtime(1337,6,undefined,2);w.player.x=w.player.px=68;let path=[],at=0,seen=0;const checks=[];for(let f=0;f<75*60&&w.phase==='running';f++){if(f===0||f-at>=60||seen!==w.dropSpawns){const reaction=f===0||seen!==w.dropSpawns?.45:0;path=safeRoute(w.player.x,w.bodies,w.bullets,8,{phase:w.phase,phaseTime:w.phaseTime,reaction});if(!path){checks.push({time:w.time,x:w.player.x,hp:w.player.hp,reaction,route6:!!safeRoute(w.player.x,w.bodies,w.bullets,6,{phase:w.phase,phaseTime:w.phaseTime,reaction}),route4:!!safeRoute(w.player.x,w.bodies,w.bullets,4,{phase:w.phase,phaseTime:w.phaseTime,reaction}),state:JSON.parse(JSON.stringify(w,(_k,v)=>v instanceof Map?[...v]:v instanceof Set?[...v]:v))});break;}at=f;seen=w.dropSpawns;}w.step(C.step,path?.[Math.floor((f-at)/6)]??w.player.x);w.drainEffects();w.bullets=w.bullets.filter(b=>b.hostile);}result.push({version,time:w.time,hp:w.player.hp,phase:w.phase,checks});}writeFileSync('build/qa/Stage-196/heavy-route-audit.json',JSON.stringify(result,null,2));},60000);
